                        double dx=rs.getDouble(2)-x,dz=rs.getDouble(3)-z;
                        if(dx*dx+dz*dz<=r2){count++;if(count>=3)return count;}
                    }
                }
            }
            return count;
        }

        boolean upsertWatchEntry(UUID suspect,String suspectName,UUID main,String mainName,int score,String evidence,long now)throws SQLException {
            if(suspect==null||main==null||suspect.equals(main))return false;
            String existing=null;int oldScore=0;
            try(Connection c=open();PreparedStatement p=c.prepareStatement("SELECT id,score FROM wgm_watchlist WHERE suspect_uuid=? AND main_uuid=? AND active=? ORDER BY last_seen DESC")){
                p.setString(1,suspect.toString());p.setString(2,main.toString());p.setBoolean(3,true);
                try(ResultSet r=p.executeQuery()){if(r.next()){existing=r.getString(1);oldScore=r.getInt(2);}}
            }
            if(existing!=null){
                try(Connection c=open();PreparedStatement p=c.prepareStatement("UPDATE wgm_watchlist SET suspect_name=?,main_name=?,score=?,evidence=?,last_seen=? WHERE id=?")){
                    p.setString(1,limit(suspectName,64));p.setString(2,limit(mainName,64));p.setInt(3,Math.max(oldScore,score));p.setString(4,limit(evidence,950));p.setLong(5,now);p.setString(6,existing);p.executeUpdate();
                }
                return false;
            }
            try(Connection c=open();PreparedStatement p=c.prepareStatement("INSERT INTO wgm_watchlist(id,suspect_uuid,suspect_name,main_uuid,main_name,score,evidence,created_at,last_seen,active,dismissed_by,dismissed_at) VALUES(?,?,?,?,?,?,?,?,?,?,?,?)")){
                p.setString(1,UUID.randomUUID().toString());p.setString(2,suspect.toString());p.setString(3,limit(suspectName,64));p.setString(4,main.toString());p.setString(5,limit(mainName,64));p.setInt(6,score);p.setString(7,limit(evidence,950));p.setLong(8,now);p.setLong(9,now);p.setBoolean(10,true);p.setString(11,null);p.setNull(12,Types.BIGINT);p.executeUpdate();
            }
            return true;
        }
