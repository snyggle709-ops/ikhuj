        void observeHome(UUID uuid,HomePoint point,long now)throws SQLException {
            if(uuid==null||point==null)return;
            String foundId=null;
            try(Connection c=open();PreparedStatement p=c.prepareStatement("SELECT id,x,z FROM wgm_home_history WHERE uuid=? AND home_name=? AND world_name=? ORDER BY last_seen DESC")){
                p.setString(1,uuid.toString());p.setString(2,point.name);p.setString(3,point.world);
                try(ResultSet r=p.executeQuery()){
                    while(r.next()){
                        double dx=r.getDouble(2)-point.x,dz=r.getDouble(3)-point.z;
                        if(dx*dx+dz*dz<=48.0*48.0){foundId=r.getString(1);break;}
                    }
                }
            }
            if(foundId!=null){
                try(Connection c=open();PreparedStatement p=c.prepareStatement("UPDATE wgm_home_history SET x=?,y=?,z=?,last_seen=? WHERE id=?")){
                    p.setDouble(1,point.x);p.setDouble(2,point.y);p.setDouble(3,point.z);p.setLong(4,now);p.setString(5,foundId);p.executeUpdate();
                }
            }else{
                try(Connection c=open();PreparedStatement p=c.prepareStatement("INSERT INTO wgm_home_history(id,uuid,home_name,world_name,x,y,z,first_seen,last_seen) VALUES(?,?,?,?,?,?,?,?,?)")){
                    p.setString(1,UUID.randomUUID().toString());p.setString(2,uuid.toString());p.setString(3,point.name);p.setString(4,point.world);p.setDouble(5,point.x);p.setDouble(6,point.y);p.setDouble(7,point.z);p.setLong(8,now);p.setLong(9,now);p.executeUpdate();
                }
            }
        }

        List<BasePoint> behaviorAnchors(UUID uuid,Set<String> excludedWorlds)throws SQLException {
            List<BasePoint> out=new ArrayList<>();
            if(uuid==null)return out;
            try(Connection c=open();PreparedStatement p=c.prepareStatement("SELECT home_name,world_name,x,y,z,last_seen FROM wgm_home_history WHERE uuid=? ORDER BY last_seen DESC")){
                p.setString(1,uuid.toString());
                try(ResultSet r=p.executeQuery()){
                    while(r.next()&&out.size()<8){
                        String world=r.getString(2);
                        if(excludedWorlds!=null&&excludedWorlds.contains(world.toLowerCase(Locale.ROOT)))continue;
                        BasePoint candidate=new BasePoint("HOME",world,r.getDouble(3),r.getDouble(4),r.getDouble(5),r.getLong(6));
                        boolean duplicate=false;
                        for(BasePoint existing:out)if(existing.world.equalsIgnoreCase(candidate.world)&&horizontalDistance(existing.x,existing.z,candidate.x,candidate.z)<32){duplicate=true;break;}
                        if(!duplicate)out.add(candidate);
                    }
                }
            }

            boolean hasHome=!out.isEmpty();
            if(!hasHome){
                try(Connection c=open();PreparedStatement p=c.prepareStatement("SELECT world_name,cell_x,cell_z,seconds_total,last_seen FROM wgm_presence_heat WHERE uuid=? AND seconds_total>=? ORDER BY seconds_total DESC,last_seen DESC")){
                    p.setString(1,uuid.toString());p.setLong(2,30L*60L);
                    try(ResultSet r=p.executeQuery()){
                        if(r.next()){
                            String world=r.getString(1);
                            if(excludedWorlds==null||!excludedWorlds.contains(world.toLowerCase(Locale.ROOT))){
                                int cellX=r.getInt(2),cellZ=r.getInt(3);
                                out.add(new BasePoint("HEAT",world,cellX*64.0+32.0,64.0,cellZ*64.0+32.0,r.getLong(4)));
                            }
                        }
                    }
                }
            }

            try(Connection c=open();PreparedStatement p=c.prepareStatement("SELECT world_name,x,y,z,updated_at FROM wgm_last_position WHERE uuid=?")){
                p.setString(1,uuid.toString());
                try(ResultSet r=p.executeQuery()){
                    if(r.next()){
                        String world=r.getString(1);
                        if(excludedWorlds==null||!excludedWorlds.contains(world.toLowerCase(Locale.ROOT))){
                            BasePoint last=new BasePoint("LAST",world,r.getDouble(2),r.getDouble(3),r.getDouble(4),r.getLong(5));
                            boolean duplicate=false;
                            for(BasePoint existing:out)if(existing.world.equalsIgnoreCase(last.world)&&horizontalDistance(existing.x,existing.z,last.x,last.z)<24){duplicate=true;break;}
                            if(!duplicate)out.add(last);
                        }
                    }
                }
            }
            return out;
        }
