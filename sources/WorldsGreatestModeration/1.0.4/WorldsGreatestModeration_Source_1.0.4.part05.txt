    private void openTarget(Player player, String action, int page) {
        Inventory inv = Bukkit.createInventory(null, 54, UI_TARGET);
        fill(inv, Material.GRAY_STAINED_GLASS_PANE, " ");
        List<Player> online = new ArrayList<>();
        for (Player p : Bukkit.getOnlinePlayers()) online.add(p);
        online.sort(Comparator.comparing(Player::getName, String.CASE_INSENSITIVE_ORDER));
        List<String> targetNames = new ArrayList<>(); for (Player p : online) targetNames.add(p.getName());
        targetCache.put(player.getUniqueId(), targetNames);
        int from = page * 36;
        for (int i = 0; i < 36 && from + i < online.size(); i++) {
            Player target = online.get(from + i);
            int slot = targetSlot(i);
            ItemStack head = item(Material.PLAYER_HEAD, "§f§l" + target.getName(), "§7Click to select", "§8Action: §f" + actionLabel(action));
            try {
                ItemMeta meta = head.getItemMeta();
                if (meta instanceof SkullMeta skull) { skull.setOwningPlayer(target); head.setItemMeta(skull); }
            } catch (Throwable ignored) {}
            inv.setItem(slot, head);
        }
        set(inv, 45, item(Material.OAK_SIGN, "§7Back"));
        if (page > 0) set(inv, 46, item(Material.PAPER, "§ePrevious Page"));
        set(inv, 49, item(Material.NAME_TAG, "§6§lEnter Username", "§7Click, then type any username in chat.", "§7Works for offline players too."));
        if (from + 36 < online.size()) set(inv, 52, item(Material.PAPER, "§eNext Page"));
        set(inv, 53, item(Material.BARRIER, "§cClose"));
        uiStates.put(player.getUniqueId(), new UiState("TARGET", action, null, 0L, page, null));
        player.openInventory(inv);
    }

    private int targetSlot(int i) {

    private void openHistoryAsync(Player player, String target) {
        player.sendMessage(PREFIX + "§7Loading moderation history for §f" + target + "§7...");
        UUID targetUuid = knownUuid(target);
        Bukkit.getScheduler().runTaskAsynchronously(this, () -> {
            try {
                List<Punishment> list = database.history(targetUuid, target, 250);
                historyCache.put(player.getUniqueId(), list);
                sync(() -> openHistory(player, target, 0));
            } catch (Throwable t) { asyncError(player, t); }
        });
    }

    private void openHistory(Player player, String target, int page) {
        Inventory inv = Bukkit.createInventory(null, 54, UI_HISTORY);
        fill(inv, Material.BLACK_STAINED_GLASS_PANE, " ");
        List<Punishment> list = historyCache.getOrDefault(player.getUniqueId(), Collections.emptyList());
        int from = page * 36;
        for (int i=0;i<36 && from+i<list.size();i++) {
            Punishment p = list.get(from+i);
            Material m = switch (p.type) { case "BAN" -> Material.IRON_BARS; case "BLACKLIST" -> Material.TNT; case "MUTE" -> Material.REDSTONE_TORCH; case "WARN" -> Material.YELLOW_CONCRETE; case "KICK" -> Material.FEATHER; default -> Material.PAPER; };
            List<String> lore = new ArrayList<>();
            lore.add("§7ID: §f" + p.id);
            lore.add("§7Moderator: §f" + stripColors(p.moderatorDisplay));
            lore.add("§7When: §f" + formatDate(p.createdAt));
            if (p.expiresAt == 0 && Set.of("BAN", "MUTE", "BLACKLIST", "WARN").contains(p.type)) lore.add("§7Duration: §cForever");
            else if (p.expiresAt > p.createdAt) lore.add("§7Duration: §f" + formatDuration(p.expiresAt - p.createdAt));
            lore.add("§7Status: " + (p.isActive(System.currentTimeMillis()) ? "§aACTIVE" : "§8INACTIVE"));
            lore.add("§7Reason: §f" + p.reason);
            if (p.isActive(System.currentTimeMillis()) && isModerator(player) && Set.of("BAN","MUTE","BLACKLIST","WARN").contains(p.type)) { lore.add(""); lore.add("§cRight-click to remove this action"); }
            inv.setItem(targetSlot(i), item(m, "§f§l" + p.type, lore.toArray(new String[0])));
        }
        set(inv,45,item(Material.OAK_SIGN,"§7Back"));
        if(page>0)set(inv,46,item(Material.PAPER,"§ePrevious Page"));
        set(inv,49,item(Material.NAME_TAG,"§f"+target,"§7"+list.size()+" history entries"));
        if(from+36<list.size())set(inv,52,item(Material.PAPER,"§eNext Page"));
        set(inv,53,item(Material.BARRIER,"§cClose"));
        uiStates.put(player.getUniqueId(), new UiState("HISTORY", "HISTORY", target, 0L, page, target));
        player.openInventory(inv);
    }
