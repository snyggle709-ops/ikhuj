        List<IdentityCandidate> findIdentityCandidates(UUID alt, String ip, String network, String fp, String token) throws SQLException {
            LinkedHashMap<UUID, String> found = new LinkedHashMap<>();
            if (ip != null || network != null || fp != null) {
                List<String> conditions = new ArrayList<>();
                if (ip != null) conditions.add("ip_hash=?");
                if (network != null) conditions.add("network_hash=?");
                if (fp != null) conditions.add("fingerprint_hash=?");
                String sql = "SELECT uuid,MAX(username_display) AS username_display FROM wgm_observations WHERE uuid<>? AND (" + String.join(" OR ", conditions) + ") GROUP BY uuid";
                try (Connection c = open(); PreparedStatement p = c.prepareStatement(sql)) {
                    int i = 1;
                    p.setString(i++, alt.toString());
                    if (ip != null) p.setString(i++, ip);
                    if (network != null) p.setString(i++, network);
                    if (fp != null) p.setString(i++, fp);
                    try (ResultSet r = p.executeQuery()) {
                        while (r.next()) {
                            UUID u = uuid(r.getString(1));
                            if (u != null) found.put(u, r.getString(2));
                        }
                    }
                }
            }
            if (token != null) {
                try (Connection c = open(); PreparedStatement p = c.prepareStatement("SELECT DISTINCT uuid FROM wgm_client_tokens WHERE uuid<>? AND token_hash=?")) {
                    p.setString(1, alt.toString()); p.setString(2, token);
                    try (ResultSet r = p.executeQuery()) {
                        while (r.next()) {
                            UUID u = uuid(r.getString(1));
                            if (u != null && !found.containsKey(u)) found.put(u, latestName(u));
                        }
                    }
                }
            }
            List<IdentityCandidate> out = new ArrayList<>();
            for (Map.Entry<UUID, String> e : found.entrySet()) out.add(new IdentityCandidate(e.getKey(), e.getValue()));
            return out;
        }

        Punishment resolveRootPunishment(Punishment punishment) throws SQLException {
            if (punishment == null) return null;
            Punishment current = punishment;
            Set<String> seen = new HashSet<>();
            for (int depth=0; depth<12 && current.parentId!=null && !current.parentId.isBlank() && seen.add(current.id); depth++) {
                Punishment parent = findPunishmentById(current.parentId);
                if (parent == null) break;
                current = parent;
            }
            return current;
        }
