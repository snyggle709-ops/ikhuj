    private void openWatchlistAsync(Player player, int page) {
        if (!isModerator(player)) { noPermission(player); return; }
        Bukkit.getScheduler().runTaskAsynchronously(this, () -> {
            try {
                List<WatchEntry> list = database.listWatchlist(500);
                watchlistCache.put(player.getUniqueId(), list);
                sync(() -> openWatchlist(player, list, page));
            } catch (Throwable t) {
                asyncError(player, t);
            }
        });
    }

    private void openWatchlist(Player player, List<WatchEntry> list, int page) {
        int maxPage = list.isEmpty() ? 0 : (list.size() - 1) / 36;
        page = Math.max(0, Math.min(page, maxPage));
        Inventory inv = Bukkit.createInventory(null, 54, UI_WATCHLIST);
        fill(inv, Material.GRAY_STAINED_GLASS_PANE, " ");

        int from = page * 36;
        for (int i = 0; i < 36 && from + i < list.size(); i++) {
            WatchEntry entry = list.get(from + i);
            List<String> lore = new ArrayList<>();
            lore.add("§7Possible alt of: §f" + entry.mainName);
            lore.add("§7Review score: §f" + entry.score);
            lore.add("§7First flagged: §f" + formatDate(entry.createdAt));
            lore.add("§7Last matched: §f" + formatDate(entry.lastSeen));
            lore.add("");
            lore.add("§6Evidence:");
            for (String piece : entry.evidence.split(",")) lore.add("§8• §f" + limit(piece.trim(), 46));
            lore.add("");
            lore.add("§aLeft-click: §7view suspect moderation history");
            lore.add("§cRight-click: §7dismiss from watchlist");
            lore.add("§8Review only — this entry did not punish the player.");

            ItemStack head = item(Material.PLAYER_HEAD, "§6§l" + entry.suspectName, lore.toArray(new String[0]));
            try {
                ItemMeta meta = head.getItemMeta();
                if (meta instanceof SkullMeta skull && entry.suspectUuid != null) {
                    skull.setOwningPlayer(Bukkit.getOfflinePlayer(entry.suspectUuid));
                    head.setItemMeta(skull);
                }
            } catch (Throwable ignored) {}
            inv.setItem(targetSlot(i), head);
        }

        set(inv, 45, item(Material.OAK_SIGN, "§7Back"));
        if (page > 0) set(inv, 46, item(Material.PAPER, "§ePrevious Page"));
        set(inv, 49, item(Material.SPYGLASS, "§6§lBehavioral Alt Watchlist",
                "§7Entries: §f" + list.size(),
                "§7Page: §f" + (page + 1) + "/" + (maxPage + 1),
                "",
                "§8No watchlist entry is an automatic punishment."));
        if (from + 36 < list.size()) set(inv, 52, item(Material.PAPER, "§eNext Page"));
        set(inv, 53, item(Material.BARRIER, "§cClose"));
        uiStates.put(player.getUniqueId(), new UiState("WATCHLIST", "WATCHLIST", null, 0L, page, null));
        player.openInventory(inv);
    }

    private void dismissWatchEntry(Player player, WatchEntry entry, int page) {
        Bukkit.getScheduler().runTaskAsynchronously(this, () -> {
            try {
                database.dismissWatchEntry(entry.id, player.getName(), System.currentTimeMillis());
                List<WatchEntry> fresh = database.listWatchlist(500);
                watchlistCache.put(player.getUniqueId(), fresh);
                sync(() -> openWatchlist(player, fresh, page));
            } catch (Throwable t) {
                asyncError(player, t);
            }
        });
    }

    private void sendWatchlistText(CommandSender sender) {
        Bukkit.getScheduler().runTaskAsynchronously(this, () -> {
            try {
                List<WatchEntry> list = database.listWatchlist(30);
                sync(() -> {
                    sender.sendMessage(PREFIX + "§6Behavioral alt watchlist: §f" + list.size() + (list.size() == 30 ? "+" : ""));
                    for (WatchEntry e : list) {
                        sender.sendMessage("§8- §f" + e.suspectName + " §7→ §f" + e.mainName + " §8• §6" + e.score
                                + " §8• §7" + limit(e.evidence, 120));
                    }
                    sender.sendMessage("§8Behavioral entries are review-only and do not create punishments.");
                });
            } catch (Throwable t) {
                asyncError(sender, t);
            }
        });
    }

    private void handleUiClick(Player player, UiState state, int slot, boolean rightClick) {
        switch(state.screen) {
            case "MAIN" -> {
                if(slot==49){closeUi(player);return;}
                if(slot==40 && player.isOp()){openConfig(player);return;}
                if(slot==42){openWatchlistAsync(player,0);return;}
                String action=switch(slot){case 10->"BAN";case 12->"WARN";case 14->"KICK";case 16->"MUTE";case 20->"BLACKLIST";case 22->"REMOVE";case 24,32->"HISTORY";case 30->"WARNINGS";default->null;};
                if(action!=null)openTarget(player,action,0);
            }
            case "TARGET" -> {
                if(slot==45){openMain(player);return;} if(slot==53){closeUi(player);return;}
                if(slot==46 && state.page>0){openTarget(player,state.action,state.page-1);return;}
                if(slot==52){openTarget(player,state.action,state.page+1);return;}
                if(slot==49){player.closeInventory();prompts.put(player.getUniqueId(),new Prompt("TARGET",state.action,null,0L));player.sendMessage(PREFIX+"§eType the username in chat, or type §fcancel§e.");return;}
                if(slot>=0&&slot<36){
                    List<String> names=targetCache.getOrDefault(player.getUniqueId(),Collections.emptyList());
                    int idx=state.page*36+slot; if(idx<names.size()) continueUiForTarget(player,state.action,names.get(idx));
                }
            }
            case "DURATION" -> {
                if(slot==18){openTarget(player,state.action,0);return;} if(slot==26){closeUi(player);return;}
                String raw=switch(slot){case 2->"10m";case 3->"1h";case 4->"6h";case 5->"1d";case 6->"7d";case 11->"30d";case 13->"forever";default->null;};
                if(raw!=null){openReason(player,state.action,state.target,parseDuration(raw));return;}
                if(slot==15){player.closeInventory();prompts.put(player.getUniqueId(),new Prompt("DURATION",state.action,state.target,0L));player.sendMessage(PREFIX+"§eType a duration like §f2w3d4h§e, or §fforever§e.");}
            }
            case "REASON" -> {
                if(slot==35){closeUi(player);return;} if(slot==27){ if(Set.of("BAN","MUTE","BLACKLIST","WARN").contains(state.action))openDuration(player,state.action,state.target);else openTarget(player,state.action,0);return;}
                String reason=switch(slot){case 10->"Cheating";case 11->"Exploiting";case 12->"Harassment";case 14->"Spam";case 15->"Inappropriate Behavior";case 16->"Ban Evasion";case 24->"";default->null;};
