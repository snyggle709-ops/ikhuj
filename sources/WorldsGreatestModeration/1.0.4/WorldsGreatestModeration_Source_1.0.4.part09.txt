        if (address.isLinkLocalAddress() || address.isSiteLocalAddress() || address.isMulticastAddress()) return "R" + hash.substring(1);

    private String hashNetwork(InetAddress address) {
        if (address == null) return null;
        byte[] raw = address.getAddress().clone();
        if (raw.length == 4) {
            raw[3] = 0;
        } else if (raw.length == 16) {
            for (int i = 8; i < raw.length; i++) raw[i] = 0;
        }
        String marker = address.isLoopbackAddress() || address.isAnyLocalAddress() ? "LN:" : (address.isSiteLocalAddress() || address.isLinkLocalAddress() ? "RN:" : "PN:");
        return marker + hmacBytes(raw);
    }
