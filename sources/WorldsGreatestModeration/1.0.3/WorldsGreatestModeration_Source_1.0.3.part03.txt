    private void openTarget(Player player, String action, int page) {
        Inventory inv = Bukkit.createInventory(null, 54, UI_TARGET);
        fill(inv, Material.GRAY_STAINED_GLASS_PANE, " ");
        List<Player> online = new ArrayList<>();
        for (Player p : Bukkit.getOnlinePlayers()) online.add(p);
        online.sort(Comparator.comparing(Player::getName, String.CASE_INSENSITIVE_ORDER));
        List<String> targetNames = new ArrayList<>(); for (Player p : online) targetNames.add(p.getName());
        targetCache.put(player.getUniqueId(), targetNames);
        int from = page * 36;
        for (int i = 0; i < 36 && from + i < online.size(); i++) {
            Player target = online.get(from + i);
            int slot = targetSlot(i);
            ItemStack head = item(Material.PLAYER_HEAD, "§f§l" + target.getName(), "§7Click to select", "§8Action: §f" + actionLabel(action));
            try {
                ItemMeta meta = head.getItemMeta();
                if (meta instanceof SkullMeta skull) { skull.setOwningPlayer(target); head.setItemMeta(skull); }
            } catch (Throwable ignored) {}
            inv.setItem(slot, head);
        }
        set(inv, 45, item(Material.OAK_SIGN, "§7Back"));
        if (page > 0) set(inv, 46, item(Material.PAPER, "§ePrevious Page"));
        set(inv, 49, item(Material.NAME_TAG, "§6§lEnter Username", "§7Click, then type any username in chat.", "§7Works for offline players too."));
        if (from + 36 < online.size()) set(inv, 52, item(Material.PAPER, "§eNext Page"));
        set(inv, 53, item(Material.BARRIER, "§cClose"));
        uiStates.put(player.getUniqueId(), new UiState("TARGET", action, null, 0L, page, null));
        player.openInventory(inv);
    }

        int from = page * 36;

            Material m = switch (p.type) { case "BAN" -> Material.IRON_BARS; case "BLACKLIST" -> Material.TNT; case "MUTE" -> Material.REDSTONE_TORCH; case "WARN" -> Material.YELLOW_CONCRETE; case "KICK" -> Material.FEATHER; default -> Material.PAPER; };

    private void handleUiClick(Player player, UiState state, int slot, boolean rightClick) {
        switch(state.screen) {
            case "MAIN" -> {
                if(slot==49){closeUi(player);return;}
                if(slot==40 && player.isOp()){openConfig(player);return;}
                String action=switch(slot){case 10->"BAN";case 12->"WARN";case 14->"KICK";case 16->"MUTE";case 20->"BLACKLIST";case 22->"REMOVE";case 24,32->"HISTORY";case 30->"WARNINGS";default->null;};
                if(action!=null)openTarget(player,action,0);
            }
            case "TARGET" -> {
                if(slot==45){openMain(player);return;} if(slot==53){closeUi(player);return;}
                if(slot==46 && state.page>0){openTarget(player,state.action,state.page-1);return;}
                if(slot==52){openTarget(player,state.action,state.page+1);return;}
                if(slot==49){player.closeInventory();prompts.put(player.getUniqueId(),new Prompt("TARGET",state.action,null,0L));player.sendMessage(PREFIX+"§eType the username in chat, or type §fcancel§e.");return;}
                if(slot>=0&&slot<36){
                    List<String> names=targetCache.getOrDefault(player.getUniqueId(),Collections.emptyList());
                    int idx=state.page*36+slot; if(idx<names.size()) continueUiForTarget(player,state.action,names.get(idx));
                }
            }
            case "DURATION" -> {
                if(slot==18){openTarget(player,state.action,0);return;} if(slot==26){closeUi(player);return;}
                String raw=switch(slot){case 2->"10m";case 3->"1h";case 4->"6h";case 5->"1d";case 6->"7d";case 11->"30d";case 13->"forever";default->null;};
                if(raw!=null){openReason(player,state.action,state.target,parseDuration(raw));return;}
                if(slot==15){player.closeInventory();prompts.put(player.getUniqueId(),new Prompt("DURATION",state.action,state.target,0L));player.sendMessage(PREFIX+"§eType a duration like §f2w3d4h§e, or §fforever§e.");}
            }
            case "REASON" -> {
                if(slot==35){closeUi(player);return;} if(slot==27){ if(Set.of("BAN","MUTE","BLACKLIST","WARN").contains(state.action))openDuration(player,state.action,state.target);else openTarget(player,state.action,0);return;}
                String reason=switch(slot){case 10->"Cheating";case 11->"Exploiting";case 12->"Harassment";case 14->"Spam";case 15->"Inappropriate Behavior";case 16->"Ban Evasion";case 24->"";default->null;};
                if(reason!=null){if(settings.requireReasons&&reason.isBlank()){player.sendMessage(PREFIX+"§cA reason is required.");return;}performUiAction(player,state.action,state.target,state.duration,normalizeReason(reason));return;}
                if(slot==22){player.closeInventory();prompts.put(player.getUniqueId(),new Prompt("REASON",state.action,state.target,state.duration));player.sendMessage(PREFIX+"§eType the moderation reason in chat, or §fcancel§e.");}
            }
            case "REMOVE" -> {
                if(slot==18){openTarget(player,"REMOVE",0);return;}
                if(slot==10)submitRevokeUi(player,"BAN",state.target);
                else if(slot==12)submitRevokeUi(player,"MUTE",state.target);
                else if(slot==14)submitRevokeUi(player,"BLACKLIST",state.target);
                else if(slot==16)openWarningsAsync(player,state.target);
            }
            case "HISTORY" -> {
                if(slot==45){openMain(player);return;} if(slot==53){closeUi(player);return;}
                if(slot==46&&state.page>0){openHistory(player,state.target,state.page-1);return;} if(slot==52){openHistory(player,state.target,state.page+1);return;}
                if(slot>=0&&slot<36&&rightClick){List<Punishment> list=historyCache.getOrDefault(player.getUniqueId(),Collections.emptyList());int idx=state.page*36+slot;if(idx<list.size()){Punishment p=list.get(idx);revokeHistoryItem(player,p,state.target);}}
            }
            case "WARNINGS" -> {
                if(slot==45){ if(isModerator(player))openMain(player); else closeUi(player); return;} if(slot==53){closeUi(player);return;}
                if(slot==46&&state.page>0){openWarnings(player,state.target,state.page-1);return;} if(slot==52){openWarnings(player,state.target,state.page+1);return;}
                if(slot>=0&&slot<36&&rightClick&&isModerator(player)){List<Punishment> list=warningsCache.getOrDefault(player.getUniqueId(),Collections.emptyList());int idx=state.page*36+slot;if(idx<list.size()){Punishment p=list.get(idx);UUID targetUuid=p.targetUuid;Bukkit.getScheduler().runTaskAsynchronously(this,()->{try{database.deactivatePunishmentById(p.id);List<Punishment> fresh=database.activeWarnings(targetUuid,state.target);warningsCache.put(player.getUniqueId(),fresh);sync(()->openWarnings(player,state.target,state.page));}catch(Throwable t){asyncError(player,t);}});}}
            }
            case "CONFIG" -> handleConfigClick(player,slot);
        }
    }

    private void handleConfigClick(Player player,int slot){
        if(!player.isOp()){closeUi(player);return;} if(slot==40){closeUi(player);return;}
        if(slot==28){String next=switch(settings.altMode){case "OFF"->"SAFE";case "SAFE"->"BALANCED";case "BALANCED"->"AGGRESSIVE";default->"OFF";};updateSettingUi(player,"alt_mode",next);return;}
        if(slot==30){updateSettingUi(player,"require_reasons",Boolean.toString(!settings.requireReasons));return;}
        String type=switch(slot){case 10->"CONFIG_SERVER";case 12->"CONFIG_DISCORD";case 14->"CONFIG_MODGROUPS";case 16->"CONFIG_ALERTGROUPS";case 32->"CONFIG_MUTECMDS";default->null;};
        if(type!=null){player.closeInventory();prompts.put(player.getUniqueId(),new Prompt(type,null,null,0L));player.sendMessage(PREFIX+switch(type){case "CONFIG_SERVER"->"§eType the server/network name.";case "CONFIG_DISCORD"->"§eType the Discord appeal URL.";case "CONFIG_MODGROUPS"->"§eType comma-separated LuckPerms moderator group names.";case "CONFIG_ALERTGROUPS"->"§eType comma-separated LuckPerms groups that receive alt alerts.";default->"§eType comma-separated command names muted players cannot use.";});}
    }

    private void updateSettingUi(Player player,String key,String value){Bukkit.getScheduler().runTaskAsynchronously(this,()->{try{database.setSetting(key,value);settings=Settings.load(database);sync(()->openConfig(player));}catch(Throwable t){asyncError(player,t);}});}

    private void continueUiForTarget(Player player,String action,String target){
        target=normalizeTarget(target);
        if(Set.of("BAN","MUTE","BLACKLIST","WARN").contains(action)){openDuration(player,action,target);return;}
        if(action.equals("HISTORY")){openHistoryAsync(player,target);return;}
        if(action.equals("WARNINGS")){openWarningsAsync(player,target);return;}
        if(action.equals("REMOVE")){openRemoveMenu(player,target);return;}
