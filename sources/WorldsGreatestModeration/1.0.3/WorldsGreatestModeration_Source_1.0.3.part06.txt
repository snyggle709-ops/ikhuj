        void observe(UUID uuid, String name, String ipHash, String networkHash, String fp) throws SQLException {
            if (uuid == null) return;
            long now = System.currentTimeMillis();
            try (Connection c = open()) {
                String where = ipHash == null ? "ip_hash IS NULL" : "ip_hash=?";
                try (PreparedStatement find = c.prepareStatement("SELECT id,fingerprint_hash,network_hash FROM wgm_observations WHERE uuid=? AND " + where + " ORDER BY last_seen DESC")) {
                    find.setString(1, uuid.toString());
                    if (ipHash != null) find.setString(2, ipHash);
                    try (ResultSet r = find.executeQuery()) {
                        if (r.next()) {
                            String id = r.getString(1);
                            String oldFp = r.getString(2);
                            String oldNetwork = r.getString(3);
                            try (PreparedStatement u = c.prepareStatement("UPDATE wgm_observations SET username_display=?,username_key=?,network_hash=?,fingerprint_hash=?,last_seen=?,visits=visits+1 WHERE id=?")) {
                                u.setString(1, name);
                                u.setString(2, name.toLowerCase(Locale.ROOT));
                                u.setString(3, networkHash == null ? oldNetwork : networkHash);
                                u.setString(4, fp == null ? oldFp : fp);
                                u.setLong(5, now);
                                u.setString(6, id);
                                u.executeUpdate();
                                return;
                            }
                        }
                    }
                }
                try (PreparedStatement q = c.prepareStatement("INSERT INTO wgm_observations(id,uuid,username_display,username_key,ip_hash,network_hash,fingerprint_hash,first_seen,last_seen,visits) VALUES(?,?,?,?,?,?,?,?,?,?)")) {
                    q.setString(1, UUID.randomUUID().toString());
                    q.setString(2, uuid.toString());
                    q.setString(3, name);
                    q.setString(4, name.toLowerCase(Locale.ROOT));
                    q.setString(5, ipHash);
                    q.setString(6, networkHash);
                    q.setString(7, fp);
                    q.setLong(8, now);
                    q.setLong(9, now);
                    q.setInt(10, 1);
                    q.executeUpdate();
                }
            }
        }

        void recordLogin(UUID uuid, String name, String ipHash, String networkHash, String fp, long createdAt, boolean allowed) throws SQLException {
            if (uuid == null) return;
            try (Connection c = open(); PreparedStatement p = c.prepareStatement("INSERT INTO wgm_logins(id,uuid,username_display,ip_hash,network_hash,fingerprint_hash,created_at,allowed) VALUES(?,?,?,?,?,?,?,?)")) {
                p.setString(1, UUID.randomUUID().toString());
                p.setString(2, uuid.toString());
                p.setString(3, name);
                p.setString(4, ipHash);
                p.setString(5, networkHash);
                p.setString(6, fp);
                p.setLong(7, createdAt);
                p.setBoolean(8, allowed);
                p.executeUpdate();
            }
        }

        void enrichRecentLogin(UUID uuid, long createdAt, String fp) throws SQLException {
            if (uuid == null || fp == null) return;
            try (Connection c = open(); PreparedStatement p = c.prepareStatement("UPDATE wgm_logins SET fingerprint_hash=? WHERE uuid=? AND created_at>=? AND created_at<=? AND fingerprint_hash IS NULL")) {
                p.setString(1, fp);
                p.setString(2, uuid.toString());
                p.setLong(3, createdAt - 5_000L);
                p.setLong(4, createdAt + 10_000L);
                p.executeUpdate();
            }
        }
