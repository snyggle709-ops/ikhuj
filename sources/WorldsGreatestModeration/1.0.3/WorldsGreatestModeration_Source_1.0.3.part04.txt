    private void resolveClientToken(Player player, java.util.function.Consumer<String> callback) {
        try {
            Class<?> keyType = Class.forName("org.bukkit.NamespacedKey");
            Object key = keyType.getMethod("fromString", String.class).invoke(null, "worldsgreatestmoderation:identity");
            if (key == null) { callback.accept(null); return; }
            Method retrieve = player.getClass().getMethod("retrieveCookie", keyType);
            Object result = retrieve.invoke(player, key);
            if (!(result instanceof java.util.concurrent.CompletableFuture<?> future)) { callback.accept(null); return; }
            final Object cookieKey = key;
            future.orTimeout(1500L, java.util.concurrent.TimeUnit.MILLISECONDS).whenComplete((raw, error) -> {
                if (error != null) { callback.accept(null); return; }
                try {
                    byte[] token = raw instanceof byte[] bytes ? bytes : null;
                    if (token == null || token.length < 16) {
                        token = new byte[32];
                        new SecureRandom().nextBytes(token);
                        final byte[] toStore = token.clone();
                        sync(() -> {
                            Player live = Bukkit.getPlayer(player.getUniqueId());
                            if (live == null) return;
                            try {
                                live.getClass().getMethod("storeCookie", keyType, byte[].class).invoke(live, cookieKey, toStore);
                            } catch (Throwable ignored) {}
                        });
                    }
                    callback.accept(hmacBytes(token));
                } catch (Throwable t) {
                    callback.accept(null);
                }
            });
        } catch (Throwable t) {
            callback.accept(null);
        }
    }

    private AddressIdentity addressIdentity(InetAddress address) {
        if (address == null) return new AddressIdentity(null, null, "UNKNOWN");
        String scope;
        if (address.isLoopbackAddress() || address.isAnyLocalAddress()) scope = "LOOPBACK";
        else if (address.isSiteLocalAddress()) scope = "PRIVATE";
        else if (address.isLinkLocalAddress()) scope = "LINK_LOCAL";
        else if (address.isMulticastAddress()) scope = "MULTICAST";
        else scope = "PUBLIC";
        return new AddressIdentity(hashIp(address), hashNetwork(address), scope);
    }

    private String hashIp(InetAddress address) {
        if (address == null) return null;
        String hash = hmac(address.getHostAddress());
        if (address.isLoopbackAddress() || address.isAnyLocalAddress()) return "L" + hash.substring(1);
        if (address.isLinkLocalAddress() || address.isSiteLocalAddress() || address.isMulticastAddress()) return "R" + hash.substring(1);
        return hash;
    }

    private String hashNetwork(InetAddress address) {
        if (address == null) return null;
        byte[] raw = address.getAddress().clone();
        if (raw.length == 4) {
            raw[3] = 0;
        } else if (raw.length == 16) {
            for (int i = 8; i < raw.length; i++) raw[i] = 0;
        }
        String marker = address.isLoopbackAddress() || address.isAnyLocalAddress() ? "LN:" : (address.isSiteLocalAddress() || address.isLinkLocalAddress() ? "RN:" : "PN:");
        return marker + hmacBytes(raw);
    }
