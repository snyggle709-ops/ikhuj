        Punishment resolveRootPunishment(Punishment punishment) throws SQLException {
            if (punishment == null) return null;
            Punishment current = punishment;
            Set<String> seen = new HashSet<>();
            for (int depth=0; depth<12 && current.parentId!=null && !current.parentId.isBlank() && seen.add(current.id); depth++) {
                Punishment parent = findPunishmentById(current.parentId);
                if (parent == null) break;
                current = parent;
            }
            return current;
        }

            return true;
        }
