        Inventory top = event.getView().getTopInventory();
        if (!(top.getHolder() instanceof GuardGui)) return;
        event.setCancelled(true);
        if (!player.isOp()) {
            player.closeInventory();
            return;
        }
        int slot = event.getRawSlot();
        if (slot < 0 || slot >= top.getSize()) return;
        play(player, "ui.button.click", 0.45f, 1.25f);
        handleGuiClick(player, slot);
    }

    @EventHandler(priority = EventPriority.HIGHEST)
    public void onGuiDrag(InventoryDragEvent event) {
        if (event.getView().getTopInventory().getHolder() instanceof GuardGui) event.setCancelled(true);
    }

    private void handleGuiClick(Player player, int slot) {
        switch (slot) {
            case 10 -> {
                settings.enforcePaperSafeSettings = !settings.enforcePaperSafeSettings;
                saveSettings();
                if (settings.enforcePaperSafeSettings) applySafePaperSettings(true);
                openGuiLater(player);
            }
            case 12 -> {
                settings.fallingBlockPortalBackstop = !settings.fallingBlockPortalBackstop;
                saveSettings();
                openGuiLater(player);
            }
            case 14 -> {
                settings.pistonBackstop = !settings.pistonBackstop;
                saveSettings();
                openGuiLater(player);
            }
            case 16 -> {
                settings.tripwireBackstop = !settings.tripwireBackstop;
                saveSettings();
                openGuiLater(player);
            }
            case 20 -> {
                settings.opAlerts = !settings.opAlerts;
                saveSettings();
                openGuiLater(player);
            }
            case 22 -> {
                PaperEditResult result = applySafePaperSettings(true);
                player.sendMessage(PREFIX + result.message);
                play(player, result.changed ? "entity.player.levelup" : "block.note_block.pling", 0.8f, result.changed ? 1.2f : 1.0f);
                openGuiLater(player);
            }
            case 24 -> {
                auditPaperSettings();
                player.sendMessage(PREFIX + "§aPaper/Leaf dupe-safety settings rechecked.");
                openGuiLater(player);
            }
            case 26 -> player.closeInventory();
            default -> { }
        }
    }

    private List<String> paperStateLore() {
        List<String> lore = new ArrayList<>();
        lore.add(stateLine("Piston duplication", paperState.piston));
        lore.add(stateLine("Unsafe End-portal teleportation", paperState.endPortal));
        lore.add(stateLine("Tripwire validation bypass", paperState.tripwire));
        lore.add(stateLine("Headless pistons", paperState.headless));
        lore.add(stateLine("Permanent-block exploits", paperState.permanentBreak));
        lore.add("");
        if (!paperState.fileFound) lore.add("§cCould not locate config/paper-global.yml.");
        else if (paperState.restartRecommended) lore.add("§eFile was hardened; restart server to apply.");
        else lore.add(paperState.allSafe() ? "§aDocumented native dupe switches are safe." : "§cOne or more unsafe switches are enabled.");
        return lore;
    }

    private String stateLine(String label, TriState value) {
        return switch (value) {
            case SAFE -> "§a✔ §7" + label + ": §aSAFE";
            case UNSAFE -> "§c✘ §7" + label + ": §cUNSAFE";
            case ABSENT -> "§8• §7" + label + ": §8not present";
            case UNKNOWN -> "§e? §7" + label + ": §eunknown";
        };
    }

    private void loadSettings() {
        settingsYaml = YamlConfiguration.loadConfiguration(settingsFile);
        settings = new Settings();
        settings.enforcePaperSafeSettings = settingsYaml.getBoolean("enforce-paper-safe-settings", true);
        settings.fallingBlockPortalBackstop = settingsYaml.getBoolean("backstops.falling-block-end-portal", true);
        settings.pistonBackstop = settingsYaml.getBoolean("backstops.piston-dupe-materials-if-native-unsafe", true);
        settings.tripwireBackstop = settingsYaml.getBoolean("backstops.tripwire-if-native-unsafe", true);
        settings.opAlerts = settingsYaml.getBoolean("operator-alerts", true);
        settings.auditIntervalTicks = Math.max(200L, settingsYaml.getLong("audit-interval-ticks", 1200L));
        saveSettings();
    }
