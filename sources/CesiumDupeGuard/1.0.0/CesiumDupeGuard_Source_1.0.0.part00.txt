    @Override
    public void onEnable() {
        if (!getDataFolder().exists() && !getDataFolder().mkdirs()) {
            getLogger().warning("Could not create CesiumDupeGuard data folder.");
        }
        settingsFile = new File(getDataFolder(), "settings.yml");
        loadSettings();
        getServer().getPluginManager().registerEvents(this, this);

        if (settings.enforcePaperSafeSettings) {
            applySafePaperSettings(false);
        } else {
            auditPaperSettings();
        }

        auditTask = getServer().getScheduler().runTaskTimer(this, () -> {
            try { auditPaperSettings(); }
            catch (Throwable t) { getLogger().log(Level.WARNING, "Paper dupe-safety audit failed", t); }
        }, 200L, Math.max(200L, settings.auditIntervalTicks));

        getLogger().info("CesiumDupeGuard enabled in conservative mode. No player punishment, no item-count heuristics, no timing-based accusations.");
    }

    @Override
    public void onDisable() {
        if (auditTask != null) {
            try { auditTask.cancel(); } catch (Throwable ignored) { }
        }
        saveSettings();
    }

    @Override
    public boolean onCommand(CommandSender sender, Command command, String label, String[] args) {
        if (!command.getName().equalsIgnoreCase("dupeguard")) return false;
        if (!(sender instanceof Player player)) {
            sender.sendMessage(PREFIX + "This control panel is GUI-only and must be opened by an operator in game.");
            return true;
        }
        if (!player.isOp()) {
            player.sendMessage(PREFIX + "§cOnly server operators can use this command.");
            play(player, "entity.villager.no", 0.8f, 0.8f);
            return true;
        }
        openGui(player);
        return true;
    }

    private void handlePiston(Object event, String kind) {
        if (!settings.pistonBackstop) return;
        // This backstop is intentionally dormant while Paper's native protection is confirmed safe.
        // It only intervenes when allow-piston-duplication is explicitly true.
        PaperState state = paperState;
        if (!state.pistonUnsafe) return;

        Object blocksObj = callQuiet(event, "getBlocks");
        if (!(blocksObj instanceof Iterable<?> blocks)) return;
        for (Object block : blocks) {
            String mat = materialName(callQuiet(block, "getType"));
            if (!isPaperPistonDupeMaterial(mat)) continue;
            callQuiet(event, "setCancelled", true);
            pistonBlocks.incrementAndGet();
            blocked("Piston " + kind + " involving " + mat + " while Paper piston-duplication is enabled");
            return;
        }
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onTripwirePlace(BlockPlaceEvent event) {
        if (!settings.tripwireBackstop || !paperState.tripwireUnsafe) return;
        Object block = callQuiet(event, "getBlockPlaced");
        String mat = materialName(callQuiet(block, "getType"));
        if (!"TRIPWIRE_HOOK".equals(mat)) return;
        callQuiet(event, "setCancelled", true);
        tripwireBlocks.incrementAndGet();
        blocked("Tripwire hook placement while Paper tripwire placement validation is disabled");
        Object player = callQuiet(event, "getPlayer");
        if (player instanceof Player p) {
            p.sendMessage(PREFIX + "§cTripwire-hook placement is temporarily blocked because the server's Paper dupe protection is disabled. An operator must re-enable the safe setting and restart.");
        }
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onEntityPortal(EntityPortalEvent event) {
        if (!settings.fallingBlockPortalBackstop) return;
        Object entity = callQuiet(event, "getEntity");
        if (entity == null || !isFallingBlock(entity)) return;
        Object to = callQuiet(event, "getTo");
        if (to == null) return;
        Object world = callQuiet(to, "getWorld");
        Object environment = callQuiet(world, "getEnvironment");
        String env = environment == null ? "" : String.valueOf(environment);
        if (!env.toUpperCase(Locale.ROOT).contains("THE_END")) return;

        // Paper already fixes this by default. This independent backstop blocks the underlying
        // falling-block End-portal transition rather than guessing whether a player is duping.
        callQuiet(event, "setCancelled", true);
        portalBlocks.incrementAndGet();
        blocked("Falling block attempted to enter an End portal");
    }
