            }
        }
        return found ? CallResult.failure(failure) : CallResult.missing();
    }

    private static Object callQuiet(Object target, String name, Object... args) {
        CallResult r = callResult(target, name, args);
        return r.ok ? r.value : null;
    }

    private static Object invokeMethodQuiet(Object target, Method method, Object... args) {
        if (target == null || method == null) return null;
        try { return method.invoke(target, args); }
        catch (IllegalAccessException e) {
            CallResult r = callResult(target, method.getName(), args);
            return r.ok ? r.value : null;
        }
        catch (InvocationTargetException | RuntimeException ignored) { return null; }
    }

    private static boolean compatible(Class<?> param, Object arg) {
        if (arg == null) return !param.isPrimitive();
        if (!param.isPrimitive()) return param.isAssignableFrom(arg.getClass());
        return (param == boolean.class && arg instanceof Boolean) ||
                (param == int.class && arg instanceof Integer) ||
                (param == long.class && arg instanceof Long) ||
                (param == double.class && arg instanceof Double) ||
                (param == float.class && arg instanceof Float) ||
                (param == byte.class && arg instanceof Byte) ||
                (param == short.class && arg instanceof Short) ||
                (param == char.class && arg instanceof Character);
    }

    private static Object enumConstant(String className, String constant) {
        try {
            Class<?> c = Class.forName(className);
            if (!c.isEnum()) return null;
            @SuppressWarnings({"unchecked", "rawtypes"})
            Object value = Enum.valueOf((Class<? extends Enum>) c.asSubclass(Enum.class), constant);
            return value;
        } catch (Throwable ignored) { return null; }
    }

    private static Inventory asInventory(Object o) { return o instanceof Inventory ? (Inventory) o : null; }
    private static UUID asUuid(Object o) { return o instanceof UUID ? (UUID) o : null; }
    private static String asString(Object o) { return o == null ? null : String.valueOf(o); }
    private static int intValue(Object o, int fallback) { return o instanceof Number ? ((Number) o).intValue() : fallback; }
    private static double doubleValue(Object o, double fallback) { return o instanceof Number ? ((Number) o).doubleValue() : fallback; }
    private static boolean booleanValue(Object o, boolean fallback) { return o instanceof Boolean ? (Boolean) o : fallback; }

    private static String safeType(ItemStack stack) {
        try { return stack == null || stack.getType() == null ? "UNKNOWN" : stack.getType().name(); }
        catch (Throwable ignored) { return "UNKNOWN"; }
    }

    private static String safeName(Player p) {
        try { return p == null ? "unknown" : p.getName(); }
        catch (Throwable ignored) { return "unknown"; }
    }

    private static String safeKey(Object keyed) {
        if (keyed == null) return "unknown";
        Object key = callQuiet(keyed, "getKey");
        if (key != null) return String.valueOf(key);
        return String.valueOf(keyed);
    }

    private static String shortReasons(Collection<String> reasons) {
        String joined = String.join(", ", reasons);
        return joined.length() <= 160 ? joined : joined.substring(0, 157) + "...";
    }

    private static String compactPath(String path) {
        return path.length() <= 80 ? path : "..." + path.substring(path.length() - 77);
    }

    private static final class CallResult {
        final boolean exists;
        final boolean ok;
        final Object value;
        final Throwable failure;
        private CallResult(boolean exists, boolean ok, Object value, Throwable failure) {
            this.exists = exists; this.ok = ok; this.value = value; this.failure = failure;
        }
        static CallResult missing() { return new CallResult(false, false, null, null); }
        static CallResult success(Object value) { return new CallResult(true, true, value, null); }
        static CallResult failure(Throwable t) { return new CallResult(true, false, null, t); }
    }

    private static final class Inspection {
        final String itemType;
        final LinkedHashSet<String> reasons = new LinkedHashSet<>();
        int serializedBytes = -1;
        Inspection(String itemType) { this.itemType = itemType; }
        void add(String reason) { if (reason != null && reasons.size() < 20) reasons.add(reason); }
        boolean dangerous() { return !reasons.isEmpty(); }
    }

        private static int positiveInt(Properties p, String key, int def) {
            try { int v = Integer.parseInt(p.getProperty(key, Integer.toString(def)).trim()); return v > 0 ? v : def; }
            catch (RuntimeException e) { return def; }
        }
        private static long positiveLong(Properties p, String key, long def) {
            try { long v = Long.parseLong(p.getProperty(key, Long.toString(def)).trim()); return v > 0 ? v : def; }
            catch (RuntimeException e) { return def; }
        }
        private static double positiveDouble(Properties p, String key, double def) {
            try { double v = Double.parseDouble(p.getProperty(key, Double.toString(def)).trim()); return Double.isFinite(v) && v > 0 ? v : def; }
            catch (RuntimeException e) { return def; }
        }
        private static boolean bool(Properties p, String key, boolean def) {
            String v = p.getProperty(key); return v == null ? def : Boolean.parseBoolean(v.trim());
        }
