/**
 * CesiumItemGuard - strict illegal/malicious item defense for Paper/Leaf 1.21.4.
 *
 * Enforcement deliberately has no permission/OP bypass. Plugin commands are hard-gated
 * to the exact player name "toes__"; that command gate is NOT an enforcement bypass.
 */
public final class CesiumItemGuard extends JavaPlugin implements Listener {
    private static final String OWNER_NAME = "toes__";
    private static final String BLACKLIST_REASON = "Item Exploits";
    private static final Object MISSING = new Object();

    private final Set<UUID> enforcementInProgress = ConcurrentHashMap.newKeySet();
    private final Map<UUID, SlotCache> scanCache = new ConcurrentHashMap<>();
    private final AtomicLong detections = new AtomicLong();
    private final AtomicLong sanitizedContainers = new AtomicLong();

    private GuardConfig cfg;
    private File detectionLog;

    @Override
    public void onEnable() {
        if (!getDataFolder().exists() && !getDataFolder().mkdirs()) {
            getLogger().warning("Could not create plugin data directory; continuing with in-memory enforcement.");
        }
        this.detectionLog = new File(getDataFolder(), "detections.log");
        this.cfg = GuardConfig.load(new File(getDataFolder(), "config.properties"), getLogger());
        disableLegacyLocalBlacklist();

        getServer().getPluginManager().registerEvents(this, this);
        long period = Math.max(1L, cfg.scanPeriodTicks);
        getServer().getScheduler().runTaskTimer(this, this::periodicScan, 20L, period);

        Plugin wgm = getServer().getPluginManager().getPlugin("WorldsGreatestModeration");
        if (wgm == null) {
            getLogger().severe("WorldsGreatestModeration was not detected. ItemGuard cannot apply moderation punishments without WGM.");
        }
        getLogger().info("CesiumItemGuard enabled. Enforcement bypasses: none. Command controller: " + OWNER_NAME + ".");
    }

    @Override
    public void onDisable() {
        try {
            getServer().getScheduler().cancelTasks(this);
        } catch (Throwable ignored) {
        }
    }

    /* ----------------------------- hard command gate ----------------------------- */

    @Override
    public boolean onCommand(CommandSender sender, Command command, String label, String[] args) {
        if (!(sender instanceof Player) || !OWNER_NAME.equalsIgnoreCase(sender.getName())) {
            sender.sendMessage("Unknown command. Type /help for help.");
            return true;
        }
        Player player = (Player) sender;
        String sub = args.length == 0 ? "status" : args[0].toLowerCase(Locale.ROOT);
        switch (sub) {
            case "status":
                player.sendMessage("§8§m--------------------------------");
                player.sendMessage("§b§lCesiumItemGuard §7status");
                player.sendMessage("§7Detections this boot: §f" + detections.get());
                player.sendMessage("§7Sanitized container items: §f" + sanitizedContainers.get());
                player.sendMessage("§7Punishment backend: §fWorldsGreatestModeration only");
                player.sendMessage("§7Deep scan interval: §f" + cfg.scanPeriodTicks + " ticks");
                player.sendMessage("§7WGM integration: " + (hasWgm() ? "§aCONNECTED" : "§cNOT FOUND"));
                player.sendMessage("§7Enforcement bypasses: §cNONE");
                player.sendMessage("§8§m--------------------------------");
                return true;
            case "reload":
                this.cfg = GuardConfig.load(new File(getDataFolder(), "config.properties"), getLogger());
                player.sendMessage("§aCesiumItemGuard configuration reloaded. Restart to change the repeating scan interval already scheduled for this boot.");
                return true;
            case "scan": {
                Player target = player;
                if (args.length >= 2) {
                    Player found = Bukkit.getPlayerExact(args[1]);
                    if (found == null) {
                        player.sendMessage("§cThat player is not online.");
                        return true;
                    }
                    target = found;
                }
                boolean hit = scanPlayer(target, "manual-scan", true);
                player.sendMessage(hit ? "§cExploit item detected and enforcement triggered." : "§aNo exploit items detected in " + target.getName() + "'s inventory.");
                return true;
            }
            case "inspect": {
                Object hand = callQuiet(player.getInventory(), "getItemInMainHand");
                if (!(hand instanceof ItemStack)) {
                    player.sendMessage("§7No item in your main hand.");
                    return true;
                }
                Inspection inspection = inspect((ItemStack) hand);
                if (!inspection.dangerous()) {
                    player.sendMessage("§aHeld item passed all enabled checks. Serialized size: §f" + inspection.serializedBytes + " bytes§a.");
                } else {
                    player.sendMessage("§cHeld item is prohibited: §f" + String.join("; ", inspection.reasons));
                    removeMatchingDangerousItems(player, "inspect");
                    enforce(player, inspection, "manual-inspect");
                }
                return true;
            }
            case "help":
            default:
                player.sendMessage("§b/itemguard status §7- current guard status");
                player.sendMessage("§b/itemguard scan [player] §7- force a deep inventory scan");
                player.sendMessage("§b/itemguard inspect §7- inspect your main-hand item");
                player.sendMessage("§b/itemguard reload §7- reload thresholds");
                return true;
        }
    }

    @Override
    public List<String> onTabComplete(CommandSender sender, Command command, String alias, String[] args) {
        if (!(sender instanceof Player) || !OWNER_NAME.equalsIgnoreCase(sender.getName())) return Collections.emptyList();
        if (args.length == 1) {
            List<String> out = new ArrayList<>();
            String p = args[0].toLowerCase(Locale.ROOT);
            for (String s : new String[]{"status", "scan", "inspect", "reload", "help"}) if (s.startsWith(p)) out.add(s);
            return out;
        }
        if (args.length == 2 && "scan".equalsIgnoreCase(args[0])) {
            List<String> out = new ArrayList<>();
            for (Player p : Bukkit.getOnlinePlayers()) if (p.getName().toLowerCase(Locale.ROOT).startsWith(args[1].toLowerCase(Locale.ROOT))) out.add(p.getName());
            return out;
        }
        return Collections.emptyList();
    }

    /* ----------------------------- event interception ----------------------------- */

    @EventHandler(priority = EventPriority.MONITOR)
    public void onJoin(PlayerJoinEvent event) {
        Player p = asPlayer(callQuiet(event, "getPlayer"));
        if (p != null) scheduleScan(p, "join", true, 1L);
    }

    @EventHandler(priority = EventPriority.MONITOR)
    public void onCommandPreprocess(PlayerCommandPreprocessEvent event) {
        Player p = asPlayer(callQuiet(event, "getPlayer"));
        if (p != null) scheduleScan(p, "command", true, 1L);
    }

    @EventHandler(priority = EventPriority.LOWEST)
    public void onCreative(InventoryCreativeEvent event) {
        Player p = asPlayer(callQuiet(event, "getWhoClicked"));
        ItemStack cursor = asItem(callQuiet(event, "getCursor"));
        if (p != null && cursor != null) {
            Inspection result = inspect(cursor);
            if (result.dangerous()) {
                callQuiet(event, "setCancelled", true);
                callQuiet(event, "setCursor", (Object) null);
                enforce(p, result, "creative-injection");
                return;
            }
        }
        if (p != null) scheduleScan(p, "creative", true, 1L);
    }

    @EventHandler(priority = EventPriority.MONITOR)
    public void onInventoryClick(InventoryClickEvent event) {
        Player p = asPlayer(callQuiet(event, "getWhoClicked"));
        if (p != null) scheduleScan(p, "inventory-click", true, 1L);
    }

    @EventHandler(priority = EventPriority.MONITOR)
    public void onInventoryDrag(InventoryDragEvent event) {
        Player p = asPlayer(callQuiet(event, "getWhoClicked"));
        if (p != null) scheduleScan(p, "inventory-drag", true, 1L);
    }

    @EventHandler(priority = EventPriority.LOWEST)
    public void onInventoryOpen(InventoryOpenEvent event) {
        Player p = asPlayer(callQuiet(event, "getPlayer"));
        if (p != null) scanPlayer(p, "inventory-open", true);
        Inventory top = asInventory(callQuiet(event, "getInventory"));
        if (top != null) sanitizeUnattributedContainer(top, p == null ? "unknown-opener" : p.getName());
    }

    @EventHandler(priority = EventPriority.MONITOR)
    public void onInventoryClose(InventoryCloseEvent event) {
        Player p = asPlayer(callQuiet(event, "getPlayer"));
        if (p != null) scheduleScan(p, "inventory-close", true, 1L);
    }
