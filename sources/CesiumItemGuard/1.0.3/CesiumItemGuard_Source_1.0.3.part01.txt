    @EventHandler(priority = EventPriority.LOWEST)
    public void onDeath(PlayerDeathEvent event) {
        Player p = asPlayer(callQuiet(event, "getEntity"));
        Object dropsObj = callQuiet(event, "getDrops");
        if (p == null || !(dropsObj instanceof List<?>)) return;
        Inspection first = null;
        List<?> drops = (List<?>) dropsObj;
        for (int i = drops.size() - 1; i >= 0; i--) {
            Object o = drops.get(i);
            if (!(o instanceof ItemStack)) continue;
            Inspection result = inspect((ItemStack) o);
            if (result.dangerous()) {
                if (first == null) first = result;
                try { drops.remove(i); } catch (Throwable ignored) { }
            }
        }
        if (first != null) {
            removeMatchingDangerousItems(p, "death");
            enforce(p, first, "death-drops");
        }
    }

    /* ----------------------------- scanner/enforcement ----------------------------- */

    private void periodicScan() {
        List<Player> players = new ArrayList<>(Bukkit.getOnlinePlayers());
        if (players.isEmpty()) {
            periodicCursor = 0;
            return;
        }
        int budget = Math.max(1, Math.min(cfg.periodicPlayersPerCycle, players.size()));
        for (int i = 0; i < budget; i++) {
            if (periodicCursor >= players.size()) periodicCursor = 0;
            Player p = players.get(periodicCursor++);
            try {
                scanPlayer(p, "periodic", false);
            } catch (Throwable t) {
                getLogger().log(Level.WARNING, "Periodic item scan failed for " + safeName(p), t);
            }
        }
    }

    private void scheduleScan(Player p, String source, boolean force, long delay) {
        if (p == null) return;
        scanRequests.incrementAndGet();
        UUID uuid = p.getUniqueId();
        PendingScan created = new PendingScan(force, source);
        PendingScan existing = pendingScans.putIfAbsent(uuid, created);
        if (existing != null) {
            if (force) existing.force = true;
            existing.source = source;
            coalescedScanRequests.incrementAndGet();
            return;
        }
        getServer().getScheduler().runTaskLater(this, () -> {
            PendingScan pending = pendingScans.remove(uuid);
            if (pending == null) return;
            try {
                scanPlayer(p, pending.source, pending.force);
            } catch (Throwable t) {
                getLogger().log(Level.WARNING, "Scheduled item scan failed for " + safeName(p), t);
            }
        }, Math.max(1L, delay));
    }

    private boolean scanPlayer(Player player, String source, boolean force) {
        if (player == null || enforcementInProgress.contains(player.getUniqueId())) return false;
        playerScans.incrementAndGet();
        Inventory inv = player.getInventory();
        int size = inv.getSize();
        SlotCache cache = scanCache.computeIfAbsent(player.getUniqueId(), id -> new SlotCache(size));
        if (cache.hashes.length != size) cache = new SlotCache(size);
        scanCache.put(player.getUniqueId(), cache);

        long now = System.nanoTime();
        long forceEveryNanos = Math.max(1L, cfg.forceRescanTicks) * 50_000_000L;
        boolean effectiveForce = force || cache.lastForcedScanNanos == 0L || now - cache.lastForcedScanNanos >= forceEveryNanos;
        if (effectiveForce) cache.lastForcedScanNanos = now;

        Inspection first = null;
        for (int slot = 0; slot < size; slot++) {
            slotsVisited.incrementAndGet();
            ItemStack stack = inv.getItem(slot);
            int hash = stack == null ? 0 : stack.hashCode();
            if (!effectiveForce && cache.hashes[slot] == hash) continue;
            cache.hashes[slot] = hash;
            if (stack == null || stack.getType() == Material.AIR) continue;
            Inspection result = inspect(stack);
            if (result.dangerous()) {
                inv.setItem(slot, null);
                cache.hashes[slot] = 0;
                if (first == null) first = result;
                logDetection(player, result, source + ":slot=" + slot);
            }
        }

        Object open = callQuiet(player, "getOpenInventory");
        ItemStack cursor = asItem(callQuiet(open, "getCursor"));
        if (cursor != null && cursor.getType() != Material.AIR) {
            Inspection result = inspect(cursor);
            if (result.dangerous()) {
                callQuiet(open, "setCursor", (Object) null);
                if (first == null) first = result;
                logDetection(player, result, source + ":cursor");
            }
        }

        if (first != null) {
            callQuiet(player, "updateInventory");
            enforce(player, first, source);
            return true;
        }
        return false;
    }

    private void removeMatchingDangerousItems(Player player, String source) {
        if (player == null) return;
        Inventory inv = player.getInventory();
        for (int i = 0; i < inv.getSize(); i++) {
            ItemStack stack = inv.getItem(i);
            if (stack == null || stack.getType() == Material.AIR) continue;
            Inspection result = inspect(stack);
            if (result.dangerous()) {
                inv.setItem(i, null);
                logDetection(player, result, source + ":purge-slot=" + i);
            }
        }
        Object open = callQuiet(player, "getOpenInventory");
        ItemStack cursor = asItem(callQuiet(open, "getCursor"));
        if (cursor != null && inspect(cursor).dangerous()) callQuiet(open, "setCursor", (Object) null);
        callQuiet(player, "updateInventory");
    }

    private void sanitizeUnattributedContainer(Inventory inventory, String opener) {
        if (!cfg.sanitizeOpenedContainers) return;
        for (int i = 0; i < inventory.getSize(); i++) {
            ItemStack stack = inventory.getItem(i);
            if (stack == null || stack.getType() == Material.AIR) continue;
            Inspection result = inspect(stack);
            if (!result.dangerous()) continue;
            inventory.setItem(i, null);
            sanitizedContainers.incrementAndGet();
            appendLog("CONTAINER_SANITIZED opener=" + opener + " slot=" + i + " item=" + result.itemType + " reasons=" + String.join("|", result.reasons));
            getLogger().warning("Removed prohibited item from opened container (opener=" + opener + ", slot=" + i + "): " + String.join("; ", result.reasons));
