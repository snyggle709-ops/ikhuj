        shopYaml.set("prepared", null);
        for (Map.Entry<String, PreparedChest> entry : preparedContainers.entrySet()) {

    private void loadShops() {
        if (shopYaml.isConfigurationSection("shops")) {
            for (String child : shopYaml.getConfigurationSection("shops").getKeys(false)) {
                String p = "shops." + child;
                try {
                    String signKey = shopYaml.getString(p + ".sign");
                    String containerKey = shopYaml.getString(p + ".container");
                    String ownerString = shopYaml.getString(p + ".owner");
                    String ownerName = shopYaml.getString(p + ".ownerName");
                    boolean admin = shopYaml.getBoolean(p + ".admin", false);
                    boolean createdByAdmin = shopYaml.getBoolean(p + ".createdByAdmin", admin);
                    String creatorString = shopYaml.getString(p + ".creator");
                    String creatorName = shopYaml.getString(p + ".creatorName");
                    int amount = shopYaml.getInt(p + ".amount");
                    double buy = shopYaml.getDouble(p + ".buy");
                    double sell = shopYaml.getDouble(p + ".sell");
                    ItemStack item = shopYaml.getItemStack(p + ".item");
                    if (item != null && item.getType() == Material.AIR) item = null;
                    if (signKey == null || containerKey == null) continue;
                    if (amount <= 0 || amount > MAX_AMOUNT) continue;
                    if (!Double.isFinite(buy) || !Double.isFinite(sell) || buy < 0.0D || sell < 0.0D || buy > MAX_PRICE || sell > MAX_PRICE || (buy <= 0.0D && sell <= 0.0D)) continue;
                    UUID owner = ownerString == null || ownerString.isBlank() ? null : UUID.fromString(ownerString);
                    UUID creator = creatorString == null || creatorString.isBlank() ? null : UUID.fromString(creatorString);
                    if (!admin && owner == null && (ownerName == null || ownerName.isBlank())) continue;
                    if (!admin && (ownerName == null || ownerName.isBlank()) && owner != null) {
                        try { ownerName = Bukkit.getOfflinePlayer(owner).getName(); } catch (Throwable ignored) {}
                    }
                    addShop(new Shop(signKey, containerKey, owner, ownerName, admin, createdByAdmin, creator, creatorName, amount, buy, sell, item));
                } catch (Exception ex) {
                    getLogger().warning("Skipped invalid shop entry " + child + ": " + ex.getMessage());
                }
            }
        }

        if (shopYaml.isConfigurationSection("prepared")) {
            for (String child : shopYaml.getConfigurationSection("prepared").getKeys(false)) {
                String p = "prepared." + child;
                try {
                    String containerKey = shopYaml.getString(p + ".container");
                    boolean admin = shopYaml.getBoolean(p + ".admin", false);
                    String ownerName = shopYaml.getString(p + ".ownerName");
                    String creatorString = shopYaml.getString(p + ".creator");
                    String creatorName = shopYaml.getString(p + ".creatorName");
                    UUID creator = creatorString == null || creatorString.isBlank() ? null : UUID.fromString(creatorString);
                    if (containerKey == null || containerKey.isBlank()) continue;
                    if (!admin && (ownerName == null || ownerName.isBlank())) continue;
                    preparedContainers.put(containerKey, new PreparedChest(admin, ownerName, creator, creatorName));
                } catch (Exception ex) {
                    getLogger().warning("Skipped invalid prepared chest entry " + child + ": " + ex.getMessage());
                }
            }
        }
    }

    private void saveRatings() {
        ratingYaml.set("ratings", null);
        for (Map.Entry<UUID, Map<UUID, Integer>> seller : ratings.entrySet()) {
            String base = "ratings." + seller.getKey();
            for (Map.Entry<UUID, Integer> vote : seller.getValue().entrySet()) {
                ratingYaml.set(base + "." + vote.getKey(), vote.getValue());
            }
        }
        File temp = new File(getDataFolder(), "ratings.yml.tmp");
        try {
            ratingYaml.save(temp);
            try {
                Files.move(temp.toPath(), ratingFile.toPath(), StandardCopyOption.REPLACE_EXISTING, StandardCopyOption.ATOMIC_MOVE);
            } catch (AtomicMoveNotSupportedException ex) {
                Files.move(temp.toPath(), ratingFile.toPath(), StandardCopyOption.REPLACE_EXISTING);
            }
        } catch (IOException e) {
            getLogger().severe("Failed to save ratings.yml: " + e.getMessage());
            if (temp.exists()) temp.delete();
        }
    }

    private void loadRatings() {
        if (!ratingYaml.isConfigurationSection("ratings")) return;
        for (String sellerKey : ratingYaml.getConfigurationSection("ratings").getKeys(false)) {
            try {
                UUID seller = UUID.fromString(sellerKey);
                if (!ratingYaml.isConfigurationSection("ratings." + sellerKey)) continue;
                Map<UUID, Integer> votes = new HashMap<>();
                for (String raterKey : ratingYaml.getConfigurationSection("ratings." + sellerKey).getKeys(false)) {
                    try {
                        UUID rater = UUID.fromString(raterKey);
                        int score = ratingYaml.getInt("ratings." + sellerKey + "." + raterKey);
                        if (score >= 1 && score <= 5 && !rater.equals(seller)) votes.put(rater, score);
                    } catch (Exception ignored) {
                    }
                }
                if (!votes.isEmpty()) ratings.put(seller, votes);
            } catch (Exception ignored) {
            }
        }
    }

    private static final class RatingSummary {

        final int count;

            if (count == 0) return "No ratings yet";

        }
