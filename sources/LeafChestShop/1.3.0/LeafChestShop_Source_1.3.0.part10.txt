nt amount, double buy, double sell, ItemStack item) {
            this.signKey = signKey;
            this.containerKey = containerKey;
            this.owner = owner;
            this.ownerName = ownerName;
            this.admin = admin;
            this.createdByAdmin = createdByAdmin;
            this.creator = creator;
            this.creatorName = creatorName;
            this.amount = amount;
            this.buy = buy;
            this.sell = sell;
            this.item = item == null ? null : item.clone();
            if (this.item != null) this.item.setAmount(1);
        }
    }

    private static final class PreparedChest {
        final boolean admin;
        final String ownerName;
        final UUID creator;
        final String creatorName;

        PreparedChest(boolean admin, String ownerName, UUID creator, String creatorName) {
            this.admin = admin;
            this.ownerName = ownerName;
            this.creator = creator;
            this.creatorName = creatorName;
        }

        boolean isCreator(Player player) {
            if (creator != null && creator.equals(player.getUniqueId())) return true;
            return creatorName != null && creatorName.equalsIgnoreCase(player.getName());
        }
    }

    private static final class PreparedMatch {
        final String containerKey;
        final PreparedChest prepared;
        PreparedMatch(String containerKey, PreparedChest prepared) {
            this.containerKey = containerKey;
            this.prepared = prepared;
        }
    }

    private static final class SpecialChestData {
        final boolean admin;
        final String ownerName;
        final UUID creator;
        final String creatorName;
        SpecialChestData(boolean admin, String ownerName, UUID creator, String creatorName) {
            this.admin = admin;
            this.ownerName = ownerName;
            this.creator = creator;
            this.creatorName = creatorName;
        }
    }
