    @Override
    public void onEnable() {
        shopFile = new File(getDataFolder(), "shops.yml");
        if (!getDataFolder().exists()) {
            getDataFolder().mkdirs();
        }
        shopYaml = YamlConfiguration.loadConfiguration(shopFile);
        ratingFile = new File(getDataFolder(), "ratings.yml");
        ratingYaml = YamlConfiguration.loadConfiguration(ratingFile);
        loadShops();
        loadRatings();
        getServer().getPluginManager().registerEvents(this, this);
        getLogger().info("LeafChestShop enabled with " + shops.size() + " shop(s). Economy will be resolved through Vault on demand.");
    }

    @Override
    public void onDisable() {
        saveShops();
        saveRatings();
        shops.clear();
        containerIndex.clear();
        shopLocks.clear();
        ratings.clear();
        pendingEdits.clear();
        preparedContainers.clear();
        adminGuiPages.clear();
        economy = null;
    }

    @Override
    public boolean onCommand(CommandSender sender, Command command, String label, String[] args) {
        String name = command.getName();
        if (name.equalsIgnoreCase("viewadminchestshops")) {
            return handleViewAdminShops(sender);
        }
        if (name.equalsIgnoreCase("adminchestshop")) {
            return handleAdminChestShop(sender);
        }
        if (name.equalsIgnoreCase("chestshop")) {
            return handleChestShopCommand(sender, args);
        }
        if (name.equalsIgnoreCase("rate")) {
            return handleRate(sender, args);
        }
        if (name.equalsIgnoreCase("rating")) {
            return handleRating(sender, args);
        }
        if (!name.equalsIgnoreCase("leafchestshop") && !name.equalsIgnoreCase("lcs")) {
            return false;
        }
        if (args.length == 0 || args[0].equalsIgnoreCase("help")) {
            sender.sendMessage(PREFIX + "Create a wall sign attached to a chest, trapped chest, or barrel:");
            sender.sendMessage("§a[Shop]§r   (or §c[AdminShop]§r with permission)");
            sender.sendMessage("§f<amount>§r");
            sender.sendMessage("§fB <shopBuyPrice> S <shopSellPrice>§r  (either side may be omitted)");
            sender.sendMessage(PREFIX + "B = the shop BUYS from players. S = the shop SELLS to players.");
            sender.sendMessage("§f<material>§r, §f?§r for held item, or leave line 4 blank and set the item later by right-clicking the sign with it.");
            sender.sendMessage(PREFIX + "One-way shops accept either left- or right-click. On BUY AND SELL shops: right-click = buy from the shop; left-click = sell to the shop.");
            sender.sendMessage(PREFIX + "Owners: sneak + right-click the sign to edit; right-click the chest to stock it; sneak + right-click the chest for detailed info.");
            sender.sendMessage(PREFIX + "Ratings: /rate <seller> <1-5> and /rating <seller>.");
            if (sender.hasPermission("leafchestshop.admin")) {
                sender.sendMessage(PREFIX + "Admin: /adminchestshop, /viewadminchestshops, /chestshop create <username>.");
            }
            return true;
        }
        if (args[0].equalsIgnoreCase("reload")) {
            if (!sender.hasPermission("leafchestshop.admin")) {
                sender.sendMessage(PREFIX + "§cNo permission.");
                return true;
            }
            shops.clear();
            containerIndex.clear();
            ratings.clear();
            pendingEdits.clear();
            preparedContainers.clear();
            adminGuiPages.clear();
            shopYaml = YamlConfiguration.loadConfiguration(shopFile);
            ratingYaml = YamlConfiguration.loadConfiguration(ratingFile);
            loadShops();
            loadRatings();
            economy = null;
            economyClass = null;
            sender.sendMessage(PREFIX + "§aReloaded " + shops.size() + " shop(s) and ratings.");
            return true;
        }
        if (args[0].equalsIgnoreCase("purge")) {
            if (!sender.hasPermission("leafchestshop.admin")) {
                sender.sendMessage(PREFIX + "§cNo permission.");
                return true;
            }
            int removed = purgeStaleShops();
            sender.sendMessage(PREFIX + "§aPurged " + removed + " stale shop record(s).");
            return true;
        }
        if (args[0].equalsIgnoreCase("info")) {
            sender.sendMessage(PREFIX + "Shops: §a" + shops.size() + "§r. Economy: §a" + economyName() + "§r.");
            return true;
        }
        sender.sendMessage(PREFIX + "Use /lcs help, /lcs info, /lcs purge, /lcs reload, /rate, or /rating.");
        return true;
    }

    private boolean handleViewAdminShops(CommandSender sender) {
        if (!(sender instanceof Player player)) {
            sender.sendMessage(PREFIX + "Only players can open the admin shop browser.");
            return true;
        }
        if (!player.hasPermission("leafchestshop.admin")) {
            msg(player, "§cNo permission.");
            return true;
        }
        openAdminShopBrowser(player, 0);
        return true;
    }

    private boolean handleAdminChestShop(CommandSender sender) {
        if (!(sender instanceof Player player)) {
            sender.sendMessage(PREFIX + "Only players can receive an admin shop chest.");
            return true;
        }
        if (!player.hasPermission("leafchestshop.admin")) {
            msg(player, "§cNo permission.");
            return true;
        }
        ItemStack chest = specialChestItem(true, null, player);
        if (!addSingleItem(player.getInventory(), chest)) {
            msg(player, "§cYour inventory is full.");
            return true;
        }
        msg(player, "§aAdmin chest shop given. Place it, attach a wall sign, and create the shop normally. Its stock will be unlimited.");
        return true;
    }

    private boolean handleChestShopCommand(CommandSender sender, String[] args) {
        if (!(sender instanceof Player player)) {
            sender.sendMessage(PREFIX + "Only players can use this command.");
            return true;
        }
        if (!player.hasPermission("leafchestshop.admin")) {
            msg(player, "§cNo permission.");
            return true;
        }
        if (args.length != 2 || !args[0].equalsIgnoreCase("create")) {
            msg(player, "§eUsage: /chestshop create <username>");
            return true;
        }
        String targetName = args[1].trim();
        if (!targetName.matches("[A-Za-z0-9_]{1,16}")) {
            msg(player, "§cUsername must be 1-16 characters using letters, numbers, or underscore.");
            return true;
        }
        ItemStack chest = specialChestItem(false, targetName, player);
        if (!addSingleItem(player.getInventory(), chest)) {
            msg(player, "§cYour inventory is full.");
            return true;
        }
        msg(player, "§aChest given. The next shop created on that chest will belong to §f" + targetName + "§a, even if they have never joined.");
        return true;
    }
